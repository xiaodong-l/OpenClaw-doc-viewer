## 🏗️ 技术设计

### 数据库设计

#### 用户表
```sql
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  username VARCHAR(50) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role VARCHAR(20) DEFAULT 'viewer', -- admin, editor, viewer
  status VARCHAR(20) DEFAULT 'active', -- active, inactive, suspended
  avatar_url VARCHAR(512),
  email_verified BOOLEAN DEFAULT FALSE,
  mfa_enabled BOOLEAN DEFAULT FALSE,
  mfa_secret VARCHAR(255),
  last_login_at TIMESTAMP,
  last_login_ip VARCHAR(45),
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  created_by UUID REFERENCES users(id),
  updated_by UUID REFERENCES users(id)
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_status ON users(status);
CREATE INDEX idx_users_role ON users(role);
```

#### 权限表
```sql
CREATE TABLE document_permissions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  document_path VARCHAR(2048) NOT NULL,
  user_id UUID REFERENCES users(id),
  role VARCHAR(20) NOT NULL, -- owner, editor, viewer
  granted_by UUID REFERENCES users(id),
  granted_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP,
  
  UNIQUE(document_path, user_id)
);

CREATE INDEX idx_doc_perms_path ON document_permissions(document_path);
CREATE INDEX idx_doc_perms_user ON document_permissions(user_id);
CREATE INDEX idx_doc_perms_role ON document_permissions(role);
```

### API 设计

#### 认证 API
```javascript
// 用户注册
POST /api/v2/auth/register
{
  "email": "user@example.com",
  "username": "username",
  "password": "password123"
}

// 用户登录
POST /api/v2/auth/login
{
  "email": "user@example.com",
  "password": "password123"
}

// 刷新 Token
POST /api/v2/auth/refresh
{
  "refreshToken": "xxx"
}

// 登出
POST /api/v2/auth/logout
```

#### 用户管理 API
```javascript
// 获取用户列表 (Admin)
GET /api/v2/users?page=1&limit=20&role=admin&status=active

// 获取用户详情
GET /api/v2/users/:id

// 创建用户 (Admin)
POST /api/v2/users
{
  "email": "user@example.com",
  "username": "username",
  "password": "password123",
  "role": "editor"
}

// 更新用户
PUT /api/v2/users/:id
{
  "username": "newname",
  "role": "admin",
  "status": "active"
}

// 删除用户 (Admin)
DELETE /api/v2/users/:id

// 激活/禁用用户
PATCH /api/v2/users/:id/status
{
  "status": "inactive"
}

// 重置密码 (Admin)
POST /api/v2/users/:id/reset-password
{
  "newPassword": "newpass123"
}
```

#### 权限管理 API
```javascript
// 获取文档权限
GET /api/v2/permissions?path=/docs/readme.md

// 设置文档权限
POST /api/v2/permissions
{
  "documentPath": "/docs/readme.md",
  "userId": "uuid",
  "role": "editor",
  "expiresAt": "2026-12-31T23:59:59Z"
}

// 更新权限
PUT /api/v2/permissions/:id
{
  "role": "viewer"
}

// 删除权限
DELETE /api/v2/permissions/:id

// 批量设置权限
POST /api/v2/permissions/batch
{
  "documentPath": "/docs/readme.md",
  "permissions": [
    { "userId": "uuid1", "role": "editor" },
    { "userId": "uuid2", "role": "viewer" }
  ]
}
```
