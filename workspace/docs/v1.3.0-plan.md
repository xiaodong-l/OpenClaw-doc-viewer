fastify.get('/api/v1/shares/:token/public', async (request, reply) => {
  const { token } = request.params
  
  const share = await sharesService.findByToken(token)
  if (!share) {
    return reply.code(404).send({ error: '分享链接不存在或已过期' })
  }
  
  // 检查有效期
  if (share.expiresAt && new Date() > new Date(share.expiresAt)) {
    return reply.code(410).send({ error: '分享链接已过期' })
  }
  
  // 增加访问计数
  await sharesService.incrementViewCount(share.id)
  
  return {
    success: true,
    data: {
      documentPath: share.documentPath,
      documentTitle: share.documentTitle,
      permissions: share.permissions,
      createdBy: share.createdByName,
      createdAt: share.createdAt
    }
  }
})

// 获取分享文档内容 - 无需认证
fastify.get('/api/v1/shares/:token/document', async (request, reply) => {
  const { token } = request.params
  const { path: filePath } = request.query
  
  const share = await sharesService.findByToken(token)
  if (!share) {
    return reply.code(404).send({ error: '分享链接不存在' })
  }
  
  // 验证文件路径是否匹配
  if (filePath !== share.documentPath) {
    return reply.code(403).send({ error: '无权访问此文档' })
  }
  
  // 读取文件内容
  const fileContent = await fs.promises.readFile(share.documentPath, 'utf-8')
  const { html, toc } = await parseMarkdownWithToc(fileContent)
  
  return {
    success: true,
    data: {
      path: share.documentPath,
      name: path.basename(share.documentPath),
      content: fileContent,
      html,
      toc
    }
  }
})

// 批量删除收藏文档
fastify.post('/api/v1/collections/:id/docs/batch-delete', {
  preHandler: [fastify.authenticate]
}, async (request, reply) => {
  const { id: collectionId } = request.params
  const { documentPaths } = request.body
  const userId = request.user.userId
  
  // 验证收藏夹所有权
  const collection = await collectionsService.findById(collectionId)
  if (!collection || collection.userId !== userId) {
    return reply.code(403).send({ error: '无权操作此收藏夹' })
  }
  
  // 批量删除
  const deleted = await collectionsService.batchRemoveDocuments(
    collectionId,
    documentPaths
  )
  
  return {
    success: true,
    data: {
      deletedCount: deleted.length,
      deletedPaths: deleted
    }
  }
})

// 批量移动收藏文档
fastify.post('/api/v1/collections/:id/docs/batch-move', {
  preHandler: [fastify.authenticate]
}, async (request, reply) => {
  const { id: collectionId } = request.params
  const { documentPaths, targetCollectionId } = request.body
  const userId = request.user.userId
  
  // 验证两个收藏夹的所有权
  const [sourceCollection, targetCollection] = await Promise.all([
    collectionsService.findById(collectionId),
    collectionsService.findById(targetCollectionId)
  ])
  
  if (!sourceCollection || sourceCollection.userId !== userId) {
    return reply.code(403).send({ error: '无权操作源收藏夹' })
  }
  
  if (!targetCollection || targetCollection.userId !== userId) {
    return reply.code(403).send({ error: '无权操作目标收藏夹' })
  }
  
  // 批量移动
  const moved = await collectionsService.batchMoveDocuments(
    collectionId,
    targetCollectionId,
    documentPaths
  )
  
  return {
    success: true,
    data: {
      movedCount: moved.length
    }
  }
})

// 批量删除收藏夹
fastify.post('/api/v1/collections/batch-delete', {
  preHandler: [fastify.authenticate]
}, async (request, reply) => {
  const { collectionIds } = request.body
  const userId = request.user.userId
  
  // 验证所有权并批量删除
  const deleted = await collectionsService.batchDeleteCollections(
    collectionIds,
    userId
  )
  
  return {
    success: true,
    data: {
      deletedCount: deleted.length
    }
  }
})

#### 后端服务 (services/commentService.js)

```javascript
import fs from 'fs/promises'
import path from 'path'
import { fileURLToPath } from 'url'

const __dirname = path.dirname(fileURLToPath(import.meta.url))
const DATA_FILE = path.join(__dirname, '../data/comments.json')

class CommentService {
  async loadData() {
    try {
      const data = await fs.readFile(DATA_FILE, 'utf-8')
      return JSON.parse(data)
    } catch (e) {
      return { comments: [] }
    }
  }
  
  async saveData(data) {
    await fs.writeFile(DATA_FILE, JSON.stringify(data, null, 2))
  }
  
  async findByDocument(documentPath) {
    const data = await this.loadData()
    const comments = data.comments.filter(c => c.documentPath === documentPath)
    
    // 构建评论树
    return this.buildCommentTree(comments)
  }
  
  buildCommentTree(comments) {
    const rootComments = comments.filter(c => !c.parentId)
    const replies = comments.filter(c => c.parentId)
    
    return rootComments.map(comment => ({
      ...comment,
      replies: replies.filter(r => r.parentId === comment.id)
    }))
  }
  
  async create(comment) {
    const data = await this.loadData()
    const newComment = {
      id: `comment-${Date.now()}`,
      ...comment,
      likes: 0,
      likedBy: [],
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    }
    
    data.comments.push(newComment)
    await this.saveData(data)
    return newComment
  }
  
  async update(id, updates) {
    const data = await this.loadData()
    const index = data.comments.findIndex(c => c.id === id)
    
    if (index === -1) {
      throw new Error('评论不存在')
    }
    
    data.comments[index] = {
      ...data.comments[index],
      ...updates,
      updatedAt: new Date().toISOString()
    }
    
    await this.saveData(data)
    return data.comments[index]
  }
  
  async delete(id) {
    const data = await this.loadData()
    const index = data.comments.findIndex(c => c.id === id)
    
    if (index === -1) {
      throw new Error('评论不存在')
    }
    
    // 同时删除回复
    const toDelete = [id, ...data.comments.filter(c => c.parentId === id).map(c => c.id)]
    data.comments = data.comments.filter(c => !toDelete.includes(c.id))
    
    await this.saveData(data)
    return true
  }
  
  async like(id, userId) {
    const data = await this.loadData()
    const comment = data.comments.find(c => c.id === id)
    
    if (!comment) {
      throw new Error('评论不存在')
    }
    
    const likeIndex = comment.likedBy.indexOf(userId)
    if (likeIndex === -1) {
      comment.likedBy.push(userId)
      comment.likes++
    } else {
      comment.likedBy.splice(likeIndex, 1)
      comment.likes--
    }
    
    await this.saveData(data)
    return comment
  }
}

export default new CommentService()
```

#### 后端路由 (routes/comments.js)

```javascript
import commentService from '../services/commentService.js'

export default async function commentsRoutes(fastify, options) {
  // 获取文档评论列表
  fastify.get('/api/v1/comments', async (request, reply) => {
    const { path: documentPath } = request.query
    
    const comments = await commentService.findByDocument(documentPath)
    
    return {
      success: true,
      data: comments
    }
  })
  
  // 创建评论
  fastify.post('/api/v1/comments', {
    preHandler: [fastify.authenticate]
  }, async (request, reply) => {
    const { documentPath, content, parentId } = request.body
    const user = request.user
    
    const comment = await commentService.create({
      documentPath,
      userId: user.userId,
      userName: user.name,
      userAvatar: user.avatar,
      content,
      parentId: parentId || null
    })
    
    return {
      success: true,
      data: comment
    }
  })
  
  // 更新评论
  fastify.put('/api/v1/comments/:id', {
    preHandler: [fastify.authenticate]
  }, async (request, reply) => {
    const { id } = request.params
    const { content } = request.body
    const userId = request.user.userId
    
    const comment = await commentService.findById(id)
    if (comment.userId !== userId) {
      return reply.code(403).send({ error: '无权修改此评论' })
    }
    
    const updated = await commentService.update(id, { content })
    
    return {
      success: true,
      data: updated
    }
  })
  
  // 删除评论
  fastify.delete('/api/v1/comments/:id', {
    preHandler: [fastify.authenticate]
  }, async (request, reply) => {
    const { id } = request.params
    const userId = request.user.userId
    
    const comment = await commentService.findById(id)
    if (comment.userId !== userId) {
      return reply.code(403).send({ error: '无权删除此评论' })
    }
    
    await commentService.delete(id)
    
    return {
      success: true
    }
  })
  
  // 点赞评论
  fastify.post('/api/v1/comments/:id/like', {
    preHandler: [fastify.authenticate]
  }, async (request, reply) => {
    const { id } = request.params
    const userId = request.user.userId
    
    const comment = await commentService.like(id, userId)
    
    return {
      success: true,
      data: comment
    }
  })
}
```

<template>
  <div class="document-comments">
    <h3>💬 评论 ({{ comments.length }})</h3>
    
    <!-- 评论输入 -->
    <div class="comment-input">

        v-for="comment in comments"

        class="comment-item"

        <div class="comment-avatar">
          <el-avatar :src="comment.userAvatar || getDefaultAvatar(comment.userName)" />
        </div>
        
        <div class="comment-content">
          <div class="comment-header">
            <span class="user-name">{{ comment.userName }}</span>
            <span class="comment-date">{{ formatDate(comment.createdAt) }}</span>
          </div>
          
          <div class="comment-text">{{ comment.content }}</div>
          
          <div class="comment-actions">
            <el-button
              type="text"

              @click="handleLike(comment)"
            >
              👍 {{ comment.likes }}
            </el-button>
            
            <el-button
              type="text"
              size="small"
              @click="showReplyInput = comment.id"
            >
              回复
            </el-button>
            
            <el-button
              v-if="canDelete(comment)"
              type="text"
              size="small"
              @click="handleDelete(comment)"
            >
              删除
            </el-button>
          </div>
          
          <!-- 回复列表 -->
          <div v-if="comment.replies?.length" class="replies">
            <div
              v-for="reply in comment.replies"
              :key="reply.id"
              class="reply-item"
            >
              <!-- 回复内容 -->
            </div>
          </div>
          
          <!-- 回复输入 -->
          <el-input
            v-if="showReplyInput === comment.id"
            v-model="replyContent"
            type="textarea"
            :rows="2"
            placeholder="写下你的回复..."
            @blur="showReplyInput = null"
          >
            <template #append>
              <el-button @click="handleReply(comment.id)">发送</el-button>
            </template>
          </el-input>
        </div>
      </div>
      
      <el-empty v-if="comments.length === 0" description="暂无评论，快来抢沙发吧~" />
    </div>
  </div>

<script setup>
import { ref, onMounted, computed } from 'vue'
import { ElMessage } from 'element-plus'
import api from '@/api'
import { useUserStore } from '@/stores/user'

const props = defineProps({
  documentPath: { type: String, required: true }
})

const userStore = useUserStore()
const comments = ref([])
const newComment = ref('')
const replyContent = ref('')
const submitting = ref(false)
const showReplyInput = ref(null)

const canDelete = (comment) => {
  return userStore.user?.userId === comment.userId
}

onMounted(async () => {
  await loadComments()
})

const loadComments = async () => {
  try {
    const { data } = await api.comments.getByDocument(props.documentPath)
    comments.value = data
  } catch (e) {
    ElMessage.error('加载评论失败')
  }
}

const handleSubmit = async () => {
  if (!newComment.value.trim()) {
    ElMessage.warning('请输入评论内容')
    return
  }
  
  submitting.value = true
  try {
    await api.comments.create({
      documentPath: props.documentPath,
      content: newComment.value
    })
    
    ElMessage.success('评论成功')
    newComment.value = ''
    await loadComments()
  } catch (e) {
    ElMessage.error('评论失败')
  } finally {
    submitting.value = false
  }
}

const handleReply = async (parentId) => {
  if (!replyContent.value.trim()) {
    ElMessage.warning('请输入回复内容')
    return
  }
  
  try {
    await api.comments.create({
      documentPath: props.documentPath,
      content: replyContent.value,
      parentId
    })
    
    ElMessage.success('回复成功')
    replyContent.value = ''
    showReplyInput.value = null
    await loadComments()
  } catch (e) {
    ElMessage.error('回复失败')
  }
}

const handleLike = async (comment) => {
  try {
    await api.comments.like(comment.id)
    await loadComments()
  } catch (e) {
    ElMessage.error('点赞失败')
  }
}

const handleDelete = async (comment) => {
  try {
    await api.comments.delete(comment.id)
    ElMessage.success('删除成功')
    await loadComments()
  } catch (e) {
    ElMessage.error('删除失败')
  }
}

const formatDate = (date) => {
  return new Date(date).toLocaleString('zh-CN')
}

const getDefaultAvatar = (name) => {
  // 生成默认头像逻辑
  return null
}
</script>
